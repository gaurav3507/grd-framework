\section*{Purpose}

\emph{Revised 6 October: Lemma~\ref{lem:pooled} and Corollary~\ref{cor:pooled} close the pooled-null gap; the checks table adds the repository implementation.}

\emph{Revised 8 October, after an external claims check: Theorem~\ref{thm:consistency} now covers Benjamini--Yekutieli as well as BH, through the factor $c_m=\sum_{i\le m}1/i$ ($c_m=1$ for BH); the resolution condition is described as sufficient and not weakenable in general, rather than necessary and sufficient; the centred pooled null is approximately, not exactly, valid; and the pooled Norman screen at $\Bb=19{,}999$ (five seeds) is reported in the Remark. The lower-bound note cites \citet{perry2018optimality} as the direct precedent for the detection bound.}


The JMLR novelty check found GRD's empirical side above the journal's norm and its theory below every comparable CRL paper. The manuscript's formal results (Propositions 1 and 2, Lemmas 1 and 2, the attribution proposition) are correct but elementary. This note drafts the two results that would change that assessment.

\begin{itemize}
  \item \textbf{Theorem 3 (consistency of the verdict).} Under the paper's own model, the gate's certified set equals the set of intervened environments with probability tending to one, so $d_{\mathrm{rec}}$ is a consistent estimator of the number of identifiable latent directions, and each certified direction is recovered at rate $\sqrt{d/n}$. The resolution floor of Proposition 1 appears inside the theorem as a condition that is sufficient and cannot be weakened in general, not as a separate remark.
  \item \textbf{Theorem 2 (finite-sample validity).} With a pooled-permutation null, every gate $p$-value is exactly valid under exchangeability (approximately valid after centring), and Benjamini--Yekutieli control of the false discovery rate holds under the arbitrary dependence created by the shared control sample.
\end{itemize}

Section~\ref{sec:checks} reports numerical checks of every quantitative claim. Section~\ref{sec:gaps} lists what is not yet proved and what each result costs the paper.

\section{Setting and assumptions}

Environments $e\in\{0,1,\ldots,m\}$ produce i.i.d.\ rows $Y^{(e)}_1,\ldots,Y^{(e)}_{n_e}\in\R^d$, independent across environments, with covariance $\Sigma_e$ and precision $\Omega_e=\Sigma_e^{-1}$; $e=0$ is the control sample. Write $\Delta_e=\Omega_e-\Omega_0$ and $\widehat\Omega_e$ for the inverse sample covariance. The gate statistic is $T_e=\lambda_{\max}(\widehat\Omega_e-\widehat\Omega_0)$, its population value is $\tau_e=\lambda_{\max}(\Delta_e)$, and the $p$-value is $p_e=(1+\#\{b:T_e^{(b)}\ge T_e\})/(\Bb+1)$ over $\Bb$ null draws.

\begin{enumerate}
  \item[(A1)] \emph{Gaussian rows.} Each $Y^{(e)}_i\sim N(\mu_e,\Sigma_e)$.
  \item[(A2)] \emph{Bounded spectra.} $K^{-1}\le\lambda_{\min}(\Sigma_e)\le\lambda_{\max}(\Sigma_e)\le K$ for all $e$.
  \item[(A3)] \emph{Dichotomy.} Every environment is either null, $e\in\Hzero$ with $\Sigma_e=\Sigma_0$, or shifted, $e\in\Hone$ with $\tau_e>0$. Write $k=|\Hone|$, $m_0=|\Hzero|$, and $\gamma=\min_{e\in\Hone}\tau_e$.
  \item[(A4)] \emph{Sizes.} $n_0\ge 2\max_e n_e$, so the disjoint null of the paper applies to every environment.
\end{enumerate}

By Lemma 4 of the paper, $\tau_e>0$ if and only if $\Sigma_e\not\succeq\Sigma_0$. Environments with $\Sigma_e\succeq\Sigma_0$, $\Sigma_e\ne\Sigma_0$ (pure variance inflation) are excluded by (A3); Remark~\ref{rem:inflation} returns to them.

\section{Two lemmas}

\begin{lemma}[Precision concentration]
\label{lem:conc}
Under (A1) and (A2) there is an absolute constant $c$ such that, for $t>0$ and $n\ge c K^2(d+t)$, the sample precision of $n$ rows satisfies
\[
  \|\widehat\Omega-\Omega\|_{\op{op}}\;\le\;\varepsilon(n,t):=cK^3\sqrt{(d+t)/n}
\]
with probability at least $1-2e^{-t}$.
\end{lemma}

\begin{proof}
By the Gaussian covariance bound \citep[Thm.~5.39, Rem.~5.40]{vershynin2012nonasymptotic}, $\|\widehat\Sigma-\Sigma\|\le c_1K\sqrt{(d+t)/n}$ with probability at least $1-2e^{-t}$ (estimating the mean costs one degree of freedom and changes only $c_1$). If $\|\widehat\Sigma-\Sigma\|\le 1/(2K)$, which the condition on $n$ ensures, then $\lambda_{\min}(\widehat\Sigma)\ge 1/(2K)$ and
\[\|\widehat\Sigma^{-1}-\Sigma^{-1}\|=\|\widehat\Sigma^{-1}(\Sigma-\widehat\Sigma)\Sigma^{-1}\|\le 2K\cdot K\cdot\|\widehat\Sigma-\Sigma\|.\]
\end{proof}

\begin{lemma}[Null draws vanish]
\label{lem:null}
Let a null draw split the control sample, uniformly at random and independently of the data, into disjoint parts of sizes $n_e$ and $n_0-n_e$, and let $T^{(b)}_e$ be the gate statistic of the two parts. Under (A1), (A2), and (A4), each draw satisfies $P\bigl(T^{(b)}_e>2\varepsilon(n_e,t)\bigr)\le 4e^{-t}$. For the $(1-\alpha)$ quantile $\widehat c_e(\alpha)$ of $\Bb$ draws, $P\bigl(\widehat c_e(\alpha)>2\varepsilon(n_e,t)\bigr)\le 4e^{-t}/\alpha$.
\end{lemma}

\begin{proof}
Because the split is independent of the data and the control rows are i.i.d., the two parts are, unconditionally, independent i.i.d.\ samples from $N(\mu_0,\Sigma_0)$ of sizes $n_e$ and $n_0-n_e\ge n_e$. Weyl's inequality gives $T^{(b)}_e\le\|\widehat\Omega_A-\Omega_0\|+\|\widehat\Omega_B-\Omega_0\|$, and Lemma~\ref{lem:conc} bounds each term. For the quantile, the fraction $F$ of draws exceeding $2\varepsilon$ has $E F\le 4e^{-t}$, and $\widehat c_e(\alpha)>2\varepsilon$ requires $F>\alpha$; apply Markov's inequality.
\end{proof}

\section{Theorem 3: consistency of the admissibility verdict}

\setcounter{theorem}{2}
\begin{theorem}[Verdict consistency]
\label{thm:consistency}
Assume (A1) to (A4), let $\Chat$ be the set certified at level $q$ by BH or by Benjamini--Yekutieli, write $c_m=1$ for BH and $c_m=\sum_{i=1}^m1/i$ for Benjamini--Yekutieli, and fix $\delta\in(0,1)$. Set $t=\log\{4(m+k\Bb)/\delta\}$ and $\bar\varepsilon=\max_e\varepsilon(n_e,t)$, and assume $\min_e n_e\ge cK^2(d+t)$ as in Lemma~\ref{lem:conc}.
\begin{enumerate}
  \item[(i)] \textbf{Accurate statistics.} With probability at least $1-\delta$, $|T_e-\tau_e|\le2\bar\varepsilon$ for every $e$, and no null draw of any shifted environment exceeds $2\bar\varepsilon$.
  \item[(ii)] \textbf{Sure screening.} If $\gamma>4\bar\varepsilon$ and
  \begin{equation}
    (\Bb+1)\,k\,q\;\ge\;m\,c_m,
    \label{eq:resolution}
  \end{equation}
  then $P(\Hone\subseteq\Chat)\ge1-\delta$. Conversely, if $(\Bb+1)\,k\,q<m\,c_m$ and every null $p$-value exceeds $q/c_m$, then $\Chat=\emptyset$ however large $\gamma$ is, so condition~\eqref{eq:resolution} cannot be weakened.
  \item[(iii)] \textbf{Exact recovery.} If, in addition, the null $p$-values are valid, $P(p_e\le u)\le u$ for $e\in\Hzero$ and $u\in[0,1]$, then
  \[
    P\bigl(\Chat=\Hone\bigr)\;\ge\;1-\delta-m_0\,q/c_m .
  \]
  Hence, if $q=q_n\to0$, $\Bb\ge m\,c_m/(kq_n)$, and $\gamma/\bar\varepsilon\to\infty$, the verdict $d_{\mathrm{rec}}=\min\{|\Chat|,d_\star\}$ converges in probability to $\min\{k,d_\star\}$.
\end{enumerate}
\end{theorem}

\begin{proof}
Benjamini--Yekutieli at level $q$ is BH at level $q/c_m$, so it suffices to prove the BH case ($c_m=1$) at a generic level and substitute $q/c_m$.

(i) Apply Lemma~\ref{lem:conc} to each of the $m+1$ samples with probability at least $1-2e^{-t}$ each and Weyl's inequality, $|T_e-\tau_e|\le\|\widehat\Omega_e-\Omega_e\|+\|\widehat\Omega_0-\Omega_0\|$; apply Lemma~\ref{lem:null} to the $k\Bb$ draws of shifted environments. A union bound over at most $2(m+1)+4k\Bb\le 4(m+k\Bb)$ failure events of probability $e^{-t}$ gives $1-\delta$.

(ii) On the event of (i), every $e\in\Hone$ has $T_e\ge\gamma-2\bar\varepsilon>2\bar\varepsilon\ge T^{(b)}_e$ for every draw $b$, so $p_e=1/(\Bb+1)$. The $k$ smallest $p$-values are then all equal to $1/(\Bb+1)$, and by \eqref{eq:resolution} $p_{(k)}=1/(\Bb+1)\le kq/m$. The BH rejection count $R$ is the largest $j$ with $p_{(j)}\le jq/m$, so $R\ge k$, and BH rejects every $p$-value at most $p_{(R)}\ge1/(\Bb+1)$, which includes all of $\Hone$. For the converse, by Proposition 1 of the paper any nonempty BH set has at least $m/\{q(\Bb+1)\}>k$ elements, so it must contain a null environment; but a rejected $p$-value is at most $Rq/m\le q$, and every null $p$-value exceeds $q$. Hence $R=0$.

(iii) BH rejects $e$ only if $p_e\le Rq/m\le q$. For $e\in\Hzero$, validity gives $P(p_e\le q)\le q$, so $P(\Chat\cap\Hzero\ne\emptyset)\le m_0q$. Combine with (ii). The limit statement follows because $\delta$ can be taken to zero with $t=\log\{4(m+k\Bb)/\delta\}$ growing only logarithmically while $\gamma/\bar\varepsilon\to\infty$.
\end{proof}

Parts (i) and (ii) hold for the paper's disjoint null as it is. Part (iii) needs valid null $p$-values, which the disjoint null provides only approximately; Theorem~\ref{thm:validity} supplies them for the centred pooled-permutation null, and Lemma~\ref{lem:pooled} shows that parts (i) and (ii) also hold for that null, with $\bar\varepsilon$ replaced by a bound larger by a logarithmic factor.

\begin{remark}[What the theorem adds to Proposition 1]
Proposition 1 shows that a coarse null forces abstention. Theorem~\ref{thm:consistency}(ii) shows that the same inequality, $(\Bb+1)kq\ge m\,c_m$, is sufficient and cannot be weakened in general: once the statistic separates, the resolution of the null is the only remaining obstacle to certifying every shifted environment. That the smallest Monte Carlo $p$-value limits multiple testing is known \citep{gandy2016framework}; the theorem places it in the verdict. The Norman screen is an instance: with $m=101$, $k=1$, and $q=0.05$, BH needs $\Bb+1\ge2{,}020$, which fails at 500 draws and holds at 9{,}999; Benjamini--Yekutieli, with $c_{101}=5.20$, needs $\Bb+1\ge10{,}499$. At $\Bb=19{,}999$ with the centred pooled null (five seeds), BH selects BAK1 in every seed, and Benjamini--Yekutieli, which then rejects only a $p$-value at the floor $1/20{,}000$, selects it in two seeds: its pooled $p$-value is $5\times10^{-5}$ in two seeds and $10^{-4}$ in three, against the threshold $9.5\times10^{-5}$. Resolution is then met, and the separation condition $\gamma>4\bar\varepsilon$ is what fails in three seeds, so Norman's single direction is borderline under the procedure that carries both guarantees.
\end{remark}

\begin{corollary}[Link to identifiability]
\label{cor:ident}
Take the model of Section 2.1 of the paper, $Y^{(e)}=RZ^{(e)}+\eta^{(e)}$ with a shared invertible $R$, a shared noise covariance $\op{Cov}(\eta^{(e)})=\sigma^2I$, and environments that are either unintervened or perform one atomic hard intervention on a target $i(e)$, with $\lambda'_{i}<\lambda_i$ whenever the target has no parents. Then:
\begin{enumerate}
  \item[(a)] $\Hone$ is exactly the set of intervened environments, so under the conditions of Theorem~\ref{thm:consistency}(iii) $d_{\mathrm{rec}}\to\min\{\#\text{intervened targets},\,d_\star\}$ in probability, the number of latent directions that the recovery theorem identifies, capped at the working dimension.
  \item[(b)] For every certified $e$, with probability at least $1-\delta$, the recovered direction satisfies
  \[\sin\angle\bigl(v_{\max}(\widehat\Omega_e-\widehat\Omega_0),\,v_{\max}(\Delta_e)\bigr)\le4\bar\varepsilon/\gamma,\]
  and, when $\sigma=0$, $v_{\max}(\Delta_e)$ lies within the angle of Equation (8) of the paper from the target's unmixing row, and equals it when the target has no parents.
\end{enumerate}
\end{corollary}

\begin{proof}
(a) The latent precision changes by $\Theta_e-\Theta_0=\lambda_i'^{-1}e_ie_i^\top-\lambda_i^{-1}(e_i-b_i)(e_i-b_i)^\top$ (Proposition 2 of the paper). If $b_i\ne0$, take $v\perp e_i-b_i$ with $v^\top e_i\ne0$; then $v^\top(\Theta_e-\Theta_0)v=\lambda_i'^{-1}(v^\top e_i)^2>0$. If $b_i=0$, $\Theta_e-\Theta_0=(\lambda_i'^{-1}-\lambda_i^{-1})e_ie_i^\top$, which has a positive eigenvalue when $\lambda'_i<\lambda_i$. Either way $\Sigma_{Z,e}\not\succeq\Sigma_{Z,0}$ by Lemma 4 of the paper. Congruence by $R$ and adding the common $\sigma^2I$ preserve the Loewner relation of the two covariances, because $\Sigma_e-\Sigma_0=R(\Sigma_{Z,e}-\Sigma_{Z,0})R^\top$; so $\tau_e>0$. Unintervened environments have $\Sigma_e=\Sigma_0$. (b) For positive definite $A,B$, $A^{-1}-B^{-1}=A^{-1/2}(I-A^{1/2}B^{-1}A^{1/2})A^{-1/2}$ and $B-A=A^{1/2}(A^{-1/2}BA^{-1/2}-I)A^{1/2}$, so $A^{-1}-B^{-1}$ and $B-A$ have the same inertia. Applying this twice, $\Delta_e$ has the inertia of $\Sigma_0-\Sigma_e=R(\Sigma_{Z,0}-\Sigma_{Z,e})R^\top$, hence of $\Theta_e-\Theta_0$: rank at most two and at most one positive eigenvalue, with or without observation noise. Its eigengap satisfies $\lambda_1-\lambda_2\ge\tau_e\ge\gamma$, and the Davis--Kahan variant of \citet{yu2015davis} gives $\sin\theta\le2\|E\|/\gamma$ with $\|E\|\le2\bar\varepsilon$ on the event of Theorem~\ref{thm:consistency}(i). The noiseless statement is Proposition 2 of the paper.
\end{proof}

\begin{remark}[Off the model]
\label{rem:inflation}
Proposition 5 of the paper shows that measurement changes can also have $\tau_e>0$. Theorem~\ref{thm:consistency} does not distinguish them: off the model, $d_{\mathrm{rec}}$ is a consistent estimator of the number of \emph{detectable} directions, and equals the number of identifiable directions only under the corollary's model. Environments with pure variance inflation, $\Sigma_e\succeq\Sigma_0$, have $\tau_e\le0$ and are not certified with probability tending to one; they are exactly the invisible measurement changes of Proposition 5(i).
\end{remark}

\section{Theorem 2: exact validity with a pooled-permutation null}

The paper's disjoint null draws both pseudo-samples from controls. The pooled-permutation null instead pools $Y^{(e)}$ and $Y^{(0)}$, relabels the $n_e+n_0$ rows uniformly at random into groups of sizes $n_e$ and $n_0$, and recomputes the statistic. Because Perturb-seq perturbations often shift means, each group is first centred at its own mean; Section~\ref{sec:checks} shows why this matters.

\setcounter{theorem}{1}
\begin{theorem}[Exact validity and dependence-robust FDR]
\label{thm:validity}
Suppose that for every $e\in\Hzero$ the rows of $Y^{(e)}$ and $Y^{(0)}$ are jointly exchangeable (for example, i.i.d.\ from one distribution). With pooled-permutation $p$-values:
\begin{enumerate}
  \item[(i)] $P(p_e\le u)\le u$ for every $e\in\Hzero$ and $u\in[0,1]$, for any $\Bb\ge1$ and any sample sizes;
  \item[(ii)] the Benjamini--Yekutieli procedure at level $q$ satisfies $\op{FDR}\le q\,m_0/m$, whatever the dependence among the $p$-values, including the dependence induced by the shared control sample.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) Under exchangeability, the observed labelling is one uniformly random element of the relabelling group, independent of the $\Bb$ random relabellings, so the observed statistic is exchangeable with the $\Bb$ draws; the $p$-value $(1+\#)/(\Bb+1)$ is then super-uniform \citep{phipson2010permutation,hemerik2018exact}. (ii) Benjamini--Yekutieli control holds for any joint law of valid $p$-values \citep[Theorem 1.3]{benjamini2001control}.
\end{proof}

\begin{remark}[Centring]
Exchangeability, and so exactness, holds for environments with the same distribution as the controls. An environment that shifts only its mean has $\tau_e=0$ but is not exchangeable with the controls, and without centring the pooled test rejects it: in our check, 10\% of such environments had $p\le0.05$. Centring each group at its own mean removes the shift; the centred residuals are exchangeable only approximately (each group's residuals sum to zero), so validity becomes approximate, with error of order $1/n_e$. The same check gives rejection rates of 3.3\% to 7.3\% at the 5\% level after centring (Section~\ref{sec:checks}).
\end{remark}

The pooled null keeps the consistency of Theorem~\ref{thm:consistency}. The difference from Lemma~\ref{lem:null} is that under $\Hone$ the pool mixes two distributions, so its subsets are not samples from either; the point is that both pseudo-groups are random subsets of the same pool, and therefore estimate the same matrix.

\begin{lemma}[Pooled null draws vanish, for every environment]
\label{lem:pooled}
Assume (A1), (A2), and (A4), and let $N=n_e+n_0$. There are absolute constants $c,c'$ such that, for $\eta\in(0,1)$ and $n_e\ge cK^4(d+\log(N/\eta))\log(2d/\eta)$, a centred pooled-permutation draw satisfies
\[
  P\Bigl(T^{(b)}_e>\varepsilon'(n_e,\eta)\Bigr)\le c'\eta,
  \qquad
  \varepsilon'(n_e,\eta):=cK^3\sqrt{\frac{\{d+\log(N/\eta)\}\log(2d/\eta)}{n_e}},
\]
whether $e\in\Hzero$ or $e\in\Hone$.
\end{lemma}

\begin{proof}
Condition on the data and write $z_1,\ldots,z_N$ for the centred pooled rows, $S_P=N^{-1}\sum_iz_iz_i^\top$, and $L=\max_i\|z_i\|^2$. A draw splits the pool uniformly into $A$ ($n_e$ rows) and $B$ ($n_0$ rows); let $S_A,S_B$ be the corresponding averages of $z_iz_i^\top$. Since $n_eS_A+n_0S_B=NS_P$, we have $S_B-S_P=-(n_e/n_0)(S_A-S_P)$, so it suffices to control $S_A$.

\emph{Step 1: the pool.} By Lemma~\ref{lem:conc} applied to each group and $n_0\ge2n_e$, with probability at least $1-4\eta$, $\|S_P\|\le2K$ and, since $\lambda_{\min}(\widehat\Sigma_0)\ge1/(2K)$ on the event of Lemma~\ref{lem:conc}, $\lambda_{\min}(S_P)\ge(n_0/N)\lambda_{\min}(\widehat\Sigma_0)\ge1/(3K)$. By a $\chi^2$ tail bound \citep{laurent2000adaptive} and a union bound over the $N$ rows, $L\le cK\{d+\log(N/\eta)\}$ with probability at least $1-2\eta$.

\emph{Step 2: one draw.} $S_A$ is the average of $n_e$ matrices sampled without replacement from $\{z_iz_i^\top\}$, whose mean is $S_P$. Sampling without replacement satisfies the same matrix moment-generating-function bounds as sampling with replacement \citep{gross2010note}, so matrix Bernstein \citep{tropp2012user}, with summands bounded by $2L$ and variance proxy $L\|S_P\|$, gives
$\|S_A-S_P\|\le\sqrt{2L\|S_P\|\log(2d/\eta)/n_e}+(4L/3)\log(2d/\eta)/n_e$
with probability at least $1-\eta$. The same argument on the Hermitian dilation of the subset mean $m_A$ (the pool has mean zero) gives $\|m_A\|^2\le cL\log(2d/\eta)/n_e$. The sample covariance of $A$ is $\frac{n_e}{n_e-1}(S_A-m_Am_A^\top)$, so on these events it lies within $\rho=cK\sqrt{\{d+\log(N/\eta)\}\log(2d/\eta)/n_e}$ of $S_P$, and the sample covariance of $B$ lies within $(n_e/n_0)\rho+O(\rho^2)\le\rho$ of it.

\emph{Step 3: inversion.} The condition on $n_e$ makes $\rho\le\lambda_{\min}(S_P)/2$, so both inverses lie within $2\{3K\}^2\rho$ of $S_P^{-1}$, and by Weyl's inequality $T^{(b)}_e\le\|\widehat\Omega_A-\widehat\Omega_B\|\le36K^2\rho=\varepsilon'$. No step used $\Sigma_e=\Sigma_0$: only the spectral bounds (A2) and $n_0\ge2n_e$.
\end{proof}

\begin{corollary}[Theorem~\ref{thm:consistency} under the pooled null]
\label{cor:pooled}
With the centred pooled null, parts (i) and (ii) of Theorem~\ref{thm:consistency} hold with $\bar\varepsilon$ replaced by $\bar\varepsilon'=\max_e\varepsilon'(n_e,\eta)$ for $\eta=\delta/\{c'(m+k\Bb)\}$, and part (iii) holds with the null $p$-values of Theorem~\ref{thm:validity}: exactly valid for environments with the same distribution as the controls, approximately valid after centring for environments that shift only their mean.
\end{corollary}

\begin{proof}
Replace Lemma~\ref{lem:null} by Lemma~\ref{lem:pooled} in the proof of Theorem~\ref{thm:consistency}; the union bound has the same number of terms.
\end{proof}

\section{Gaps and cost to the paper}
\label{sec:gaps}

\begin{enumerate}
  \item \textbf{Pooled-null consistency.} Closed by Lemma~\ref{lem:pooled} and Corollary~\ref{cor:pooled} (6 October). The bound is larger than the disjoint one by a factor $\sqrt{\log(2d/\eta)}$ and uses only (A2) and $n_0\ge2n_e$.
  \item \textbf{Code and reruns.} The centred pooled null and BY are in the gate (commit \texttt{8ebe72d}; default path unchanged). The A100 reruns are complete: E8 at $\Bb=9{,}999$ for the three Perturb-seq panels (\texttt{results/e8\_pooled\_null/<dataset>\_s<seed>}) and Norman at $\Bb=19{,}999$ (\texttt{norman\_B19999\_s<seed>}). The paper reports BH and BY sets for both nulls; the BH decisions with the size-matched null remain the primary real-data decision, and the BY results are reported alongside.
  \item \textbf{Assumption (A3).} The dichotomy excludes environments that shift only slightly ($0<\tau_e\le4\bar\varepsilon$). A version with three classes (null, weak, strong) and a guarantee only for the strong class is straightforward and more honest for real data.
  \item \textbf{Gaussianity.} (A1) can be relaxed to sub-Gaussian rows with the same rate; heavy-tailed single-cell data are not covered.
  \item \textbf{Novelty.} The ingredients (Weyl, covariance concentration, Davis--Kahan, randomization tests, BH and BY) are standard. What is new is the statement: a consistency guarantee for an admissibility verdict that is coupled to an identifiability theorem, with the null resolution as an explicit condition that is sufficient and cannot be weakened in general. On the rubric used for the JMLR check this moves the theory score from 2 to about 3, not to the 4 or 5 of identifiability papers.
\end{enumerate}
